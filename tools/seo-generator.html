    function generateManifest() {
    const manifest = {
    name: $("siteName").value.trim(),
    short_name:
    $("siteName").value.trim().slice(0, 12),
    start_url: "/",
    display: "browser",
    background_color:
    $("themeColor").value.trim() ||
    "#0f172a",
    theme_color:
    $("themeColor").value.trim() ||
    "#0f172a",
    description:
    $("description").value.trim(),
    scope: "/"
    };
    
    const favicon =
    $("favicon").value.trim();
    
    if (favicon) {
    manifest.icons = [
    {
    src: favicon,
    sizes: "any",
    type: "image/x-icon"
    }
    ];
    }
    
    return JSON.stringify(
    manifest,
    null,
    2
    ) + "\n";
    }

    /* =========================================================
    SEO VALIDATION
    ========================================================= */
    
    function validate() {
    const issues = [];
    
    const domain =
    normalizeDomain($("siteUrl").value);
    
    const title =
    $("title").value.trim();
    
    const description =
    $("description").value.trim();
    
    const image =
    $("ogImage").value.trim();
    
    /* Website URL */
    
    if (!domain) {
    issues.push({
    type: "error",
    text: "Website URL is missing."
    });
    } else if (!/^https:\/\//i.test(domain)) {
    issues.push({
    type: "warning",
    text: "Use HTTPS for the production website URL."
    });
    } else {
    issues.push({
    type: "good",
    text: "Website URL is configured."
    });
    }
    
    /* Title */
    
    if (!title) {
    issues.push({
    type: "error",
    text: "Homepage title is empty."
    });
    } else if (title.length < 30) { issues.push({ type: "warning" , text: `Homepage title is only ${title.length}
        characters. ` + "It may be too short to describe the page well." }); } else if (title.length> 60) {
        issues.push({
        type: "warning",
        text:
        `Homepage title is ${title.length} characters. ` +
        "Search engines may truncate it."
        });
        } else {
        issues.push({
        type: "good",
        text:
        `Homepage title length is ${title.length} characters.`
        });
        }
    
        /* Description */
    
        if (!description) {
        issues.push({
        type: "error",
        text: "Meta description is empty."
        });
        } else if (description.length < 100) { issues.push({ type: "warning" , text: `Meta description is only
            ${description.length} characters.` }); } else if (description.length> 160) {
            issues.push({
            type: "warning",
            text:
            `Meta description is ${description.length} characters. ` +
            "Search engines may truncate it."
            });
            } else {
            issues.push({
            type: "good",
            text:
            `Meta description length is ${description.length} characters.`
            });
            }
    
            /* Canonical */
    
            if (!$("canonical").value.trim() && domain) {
            issues.push({
            type: "good",
            text:
            "Canonical URL will default to the site homepage."
            });
            }
    
            /* OG image */
    
            if (!image) {
            issues.push({
            type: "warning",
            text:
            "No Open Graph image is configured. " +
            "Social shares may have no custom preview image."
            });
            } else {
            issues.push({
            type: "good",
            text: "Open Graph image is configured."
            });
            }
    
            /* Sitemap */
    
            if (pages.length === 0) {
            issues.push({
            type: "error",
            text:
            "Sitemap contains no URLs. " +
            "Add the pages you actually want indexed."
            });
            } else {
            issues.push({
            type: "good",
            text:
            `Sitemap contains ${pages.length} configured page(s).`
            });
            }
    
            /* Duplicate paths */
    
            const normalized =
            pages.map(page =>
            normalizePath(page.path)
            );
    
            const duplicates =
            normalized.filter(
            (path, index) =>
            normalized.indexOf(path) !== index
            );
    
            if (duplicates.length) {
            issues.push({
            type: "warning",
            text:
            `Duplicate sitemap path(s): ` +
            `${[...new Set(duplicates)].join(", ")}`
            });
            }
    
            /* Homepage */
    
            if (!normalized.includes("/")) {
            issues.push({
            type: "warning",
            text:
            "Your sitemap does not currently contain the homepage (/)."
            });
            }
    
            /* Noindex */
    
            if (
            $("robotsPolicy").value
            .startsWith("noindex")
            ) {
            issues.push({
            type: "warning",
            text:
            "The generated page metadata is configured with NOINDEX."
            });
            }
    
            return issues;
            }
    
            function renderValidation() {
            const issues = validate();
    
            const good =
            issues.filter(
            issue => issue.type === "good"
            ).length;
    
            const warnings =
            issues.filter(
            issue => issue.type === "warning"
            ).length;
    
            const errors =
            issues.filter(
            issue => issue.type === "error"
            ).length;
    
            $("statusGrid").innerHTML = `
            

                ${errors}
                Errors
            

    
            

                ${warnings}
                Warnings
            

    
            

                ${good}
                Checks Passed
            

    
            

                ${pages.length}
                Sitemap URLs
            

            `;
    
            $("issues").innerHTML =
            issues.map(issue => `
            

                ${
                issue.type === "good"
                ? "✓"
                : issue.type === "warning"
                ? "⚠"
                : "✕"
                }
                ${escapeHtml(issue.text)}
            

            `).join("");
            }
